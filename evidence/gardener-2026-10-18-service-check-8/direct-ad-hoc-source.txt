"""Directly invoked, offline ad-hoc verification; no project-suite claim."""
from contextlib import redirect_stdout
from datetime import datetime, timezone
from pathlib import Path
from unittest.mock import patch
import hashlib
import io
import json
import runpy
import unittest

ROOT = Path('/Users/yu/castle')
HERE = ROOT / 'evidence/gardener-2026-10-18-service-check-8'
SCRIPTS = ('collect.py', 'research.py', 'verify.py')
TRACKED = [p for p in HERE.iterdir() if p.is_file()] + [ROOT / p for p in (
    'gate.md', 'chronicle.md', 'questions.md', 'quests.md', 'courtyard.md',
    'rooms/what-the-gaps-can-say.md', 'rooms/measuring-the-window.md',
    'words/active-measurement.md')]
sha = lambda p: hashlib.sha256(p.read_bytes()).hexdigest()
snapshot = lambda: {str(p): sha(p) for p in TRACKED}
before = snapshot()
original_open, original_bytes, original_text = Path.open, Path.read_bytes, Path.read_text


def run_script(name, corruption=None):
    def open_file(path, mode='r', *args, **kwargs):
        if any(flag in mode for flag in 'wax+'):
            if path == HERE / 'verification.json' and mode == 'x':
                return io.StringIO()
            raise RuntimeError(f'Unexpected write: {path}')
        return original_open(path, mode, *args, **kwargs)

    def read_bytes(path, *args, **kwargs):
        data = original_bytes(path, *args, **kwargs)
        target = {'body': HERE / 'r1-orgmode.body',
                  'chronicle': ROOT / 'chronicle.md'}.get(corruption)
        return data + b'\nAD-HOC NEGATIVE FIXTURE\n' if path == target else data

    def read_text(path, *args, **kwargs):
        text = original_text(path, *args, **kwargs)
        if corruption == 'map' and path == ROOT / 'gate.md':
            line = next(s for s in text.splitlines() if s.startswith('- [what-the-gaps-can-say]'))
            return text.replace(line, line + '\n' + line, 1)
        return text

    output = io.StringIO()
    with patch.object(Path, 'open', open_file), \
         patch.object(Path, 'read_bytes', read_bytes), \
         patch.object(Path, 'read_text', read_text), \
         patch('subprocess.run', side_effect=RuntimeError('Network/process execution forbidden')), \
         patch('sys.argv', [str(HERE / name)]), redirect_stdout(output):
        runpy.run_path(str(HERE / name), run_name='__main__')
    return output.getvalue()


class AdHocChecks(unittest.TestCase):
    def test_three_sources_compile(self):
        for name in SCRIPTS:
            with self.subTest(script=name):
                compile((HERE / name).read_text(), str(HERE / name), 'exec')

    def test_collect_refuses_repeat_before_side_effects(self):
        with self.assertRaises(AssertionError):
            run_script('collect.py')

    def test_research_refuses_repeat_before_side_effects(self):
        with self.assertRaises(AssertionError):
            run_script('research.py')

    def test_real_verifier_passes_current_artifacts(self):
        report = json.loads(run_script('verify.py'))
        self.assertEqual(report['result'], 'PASS')
        self.assertEqual(report['chronicle_lines_added'], 1)
        self.assertTrue(report['garden_unchanged'])
        self.assertTrue(report['every_room_listed_once'])

    def test_corrupt_body_is_rejected(self):
        with self.assertRaises(AssertionError):
            run_script('verify.py', 'body')

    def test_duplicate_map_entry_is_rejected(self):
        with self.assertRaises(AssertionError):
            run_script('verify.py', 'map')

    def test_extra_chronicle_entry_is_rejected(self):
        with self.assertRaises(AssertionError):
            run_script('verify.py', 'chronicle')


if __name__ == '__main__':
    result = unittest.TextTestRunner(verbosity=2).run(
        unittest.defaultTestLoader.loadTestsFromTestCase(AdHocChecks))
    unchanged = snapshot() == before
    if not result.wasSuccessful() or not unchanged:
        raise SystemExit('Ad-hoc verification failed or existing files changed')
    print(json.dumps({
        'kind': 'ad-hoc verification, not a canonical project suite',
        'result': 'PASS', 'checks': result.testsRun,
        'utc': datetime.now(timezone.utc).isoformat(),
        'script_hashes': {name: sha(HERE / name) for name in SCRIPTS},
        'existing_files_unchanged': unchanged,
        'network_probes_repeated': 0,
        'chronicle_entries_added': 0,
        'negative_fixtures': 'in-memory only'
    }, indent=2))
